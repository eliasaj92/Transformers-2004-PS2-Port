cmake_minimum_required(VERSION 3.24)
project(tf_vu1_replay LANGUAGES CXX)

# A private, accelerated-only fixture runner. Never builds the interpreter oracle.
if(NOT MSVC OR MSVC_VERSION LESS 1930 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use Visual Studio 2022 x64 or newer")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
if(NOT PACKAGE_DIR)
    message(FATAL_ERROR "Set PACKAGE_DIR to the private accelerated VU package")
endif()
get_filename_component(PACKAGE_DIR "${PACKAGE_DIR}" REALPATH)
get_filename_component(BUILD_ROOT "${CMAKE_BINARY_DIR}" REALPATH)
string(TOLOWER "${REPO_ROOT}/" REPO_PREFIX)
foreach(PRIVATE_PATH "${PACKAGE_DIR}" "${BUILD_ROOT}")
    string(TOLOWER "${PRIVATE_PATH}/" PRIVATE_PREFIX)
    string(FIND "${PRIVATE_PREFIX}" "${REPO_PREFIX}" INSIDE_REPO)
    if(INSIDE_REPO EQUAL 0)
        message(FATAL_ERROR "Keep the package and build outside the source checkout")
    endif()
endforeach()
file(READ "${PACKAGE_DIR}/package_manifest.json" MANIFEST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${PACKAGE_DIR}/package_manifest.json" "${PACKAGE_DIR}/sources.cmake")
string(JSON ELF_SHA GET "${MANIFEST}" elf_sha256)
if(NOT ELF_SHA STREQUAL "ab88780324b9492eca59016923d200dd1d95785fd725dd85f233c071805b4acb")
    message(FATAL_ERROR "Unexpected private package identity")
endif()
string(JSON FILE_COUNT LENGTH "${MANIFEST}" files)
math(EXPR LAST_FILE "${FILE_COUNT}-1")
foreach(INDEX RANGE ${LAST_FILE})
    string(JSON REL MEMBER "${MANIFEST}" files ${INDEX})
    string(JSON EXPECTED GET "${MANIFEST}" files "${REL}")
    file(SHA256 "${PACKAGE_DIR}/${REL}" ACTUAL)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PACKAGE_DIR}/${REL}")
    if(NOT ACTUAL STREQUAL EXPECTED)
        message(FATAL_ERROR "Private package changed: ${REL}")
    endif()
endforeach()
include("${PACKAGE_DIR}/sources.cmake")
set(DUAL_STATE "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native/tf_dual_state.cpp")
option(TF_REPLAY_DUAL_TAIL "Check the short CPU tail and deferred memory against private captures" OFF)
option(TF_REPLAY_GRASS_TAIL "Check grass emission state and actual GPU outputs against private captures" OFF)
option(TF_REPLAY_REFLECTION_CAPTURE "Capture first-pass reflection inputs and outputs during accelerated replay" OFF)
option(TF_REPLAY_REFLECTION_LIVE "Check the shared live reflection hooks against enhanced captures" OFF)
if((TF_REPLAY_DUAL_TAIL AND TF_REPLAY_GRASS_TAIL) OR
   (TF_REPLAY_REFLECTION_CAPTURE AND (TF_REPLAY_DUAL_TAIL OR TF_REPLAY_GRASS_TAIL)) OR
   (TF_REPLAY_REFLECTION_LIVE AND (TF_REPLAY_DUAL_TAIL OR TF_REPLAY_GRASS_TAIL OR TF_REPLAY_REFLECTION_CAPTURE)))
    message(FATAL_ERROR "Use separate private builds for geometry diagnostics")
endif()
if(TF_REPLAY_DUAL_TAIL)
    set(REPLAY_TARGET tf_dual_tail)
    set(DUAL_MATH "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/gs/gs_native_dual_math.cpp")
    add_executable(${REPLAY_TARGET} ../native_dual_tail.cpp "${DUAL_STATE}" "${DUAL_MATH}"
        "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native/tf_dual_memory.cpp" ${PS2X_VU1_STATIC_SOURCES})
    set_source_files_properties("${DUAL_MATH}" PROPERTIES COMPILE_OPTIONS "/fp:strict")
elseif(TF_REPLAY_GRASS_TAIL)
    set(GRASS_MATH "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/gs/gs_native_grass_math.cpp")
    set(REFLECTION_MATH "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/gs/gs_native_reflection_math.cpp")
    set(DUAL_MATH "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/gs/gs_native_dual_math.cpp")
    set(GRASS_MEMORY "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native/tf_grass_memory.cpp")
    set(GRASS_RUNTIME "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native/tf_gpu_grass.cpp")
    set_source_files_properties("${GRASS_MATH}" "${DUAL_MATH}" "${REFLECTION_MATH}" PROPERTIES COMPILE_OPTIONS "/fp:strict")
    add_executable(tf_grass_memory ../native_grass_memory.cpp "${GRASS_MATH}" "${DUAL_MATH}" "${REFLECTION_MATH}" "${GRASS_MEMORY}")
    target_compile_features(tf_grass_memory PRIVATE cxx_std_20)
    target_compile_options(tf_grass_memory PRIVATE /O2 /Ob2 /fp:strict /arch:AVX2)
    target_include_directories(tf_grass_memory PRIVATE "${REPO_ROOT}/PS2Recomp/ps2xRuntime/include"
        "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS prepare_grass_chain.py)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/prepare_grass_chain.py"
        --package "${PACKAGE_DIR}" --output "${CMAKE_BINARY_DIR}/grass"
        COMMAND_ERROR_IS_FATAL ANY)
    include("${CMAKE_BINARY_DIR}/grass/grass_chain.cmake")
    list(FIND PS2X_VU1_STATIC_SOURCES "${GRASS_ORIGINAL_CHAIN}" GRASS_CHAIN_INDEX)
    if(GRASS_CHAIN_INDEX LESS 0)
        message(FATAL_ERROR "Grass chain is missing from the authenticated package source list")
    endif()
    list(REMOVE_ITEM PS2X_VU1_STATIC_SOURCES "${GRASS_ORIGINAL_CHAIN}")
    set(REPLAY_TARGET tf_grass_tail)
    add_executable(${REPLAY_TARGET} replay.cpp grass_tail.cpp "${DUAL_STATE}" "${GRASS_MATH}" "${DUAL_MATH}" "${REFLECTION_MATH}" "${GRASS_MEMORY}" "${GRASS_RUNTIME}"
        "${CMAKE_BINARY_DIR}/grass/grass_chain.cpp" ${PS2X_VU1_STATIC_SOURCES})
    target_compile_definitions(${REPLAY_TARGET} PRIVATE TF_REPLAY_GRASS_TAIL=1
        "TF_GRASS_PROLOGUE=${GRASS_PROLOGUE}" "TF_GRASS_LOOP=${GRASS_LOOP}" "TF_GRASS_EPILOGUE=${GRASS_EPILOGUE}")
    target_include_directories(${REPLAY_TARGET} PRIVATE "${PACKAGE_DIR}/generated")
elseif(TF_REPLAY_REFLECTION_CAPTURE OR TF_REPLAY_REFLECTION_LIVE)
    set(REFLECTION_MATH "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/gs/gs_native_reflection_math.cpp")
    set(DUAL_MATH "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/gs/gs_native_dual_math.cpp")
    add_executable(tf_reflection_memory ../native_reflection_memory.cpp "${REFLECTION_MATH}" "${DUAL_MATH}"
        "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native/tf_reflection_memory.cpp")
    target_compile_features(tf_reflection_memory PRIVATE cxx_std_20)
    target_compile_options(tf_reflection_memory PRIVATE /O2 /Ob2 /fp:strict /arch:AVX2)
    target_include_directories(tf_reflection_memory PRIVATE "${REPO_ROOT}/PS2Recomp/ps2xRuntime/include"
        "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    if(TF_REPLAY_REFLECTION_LIVE)
        set(REFLECTION_SCRIPT "${REPO_ROOT}/tools/vu1/prepare_reflection_hooks.py")
    else()
        set(REFLECTION_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/prepare_reflection_chain.py")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${REFLECTION_SCRIPT}")
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${REFLECTION_SCRIPT}"
        --package "${PACKAGE_DIR}" --output "${CMAKE_BINARY_DIR}/reflection"
        COMMAND_ERROR_IS_FATAL ANY)
    if(TF_REPLAY_REFLECTION_LIVE)
        include("${CMAKE_BINARY_DIR}/reflection/reflection_chains.cmake")
    else()
        include("${CMAKE_BINARY_DIR}/reflection/reflection_chain.cmake")
    endif()
    foreach(CHAIN IN LISTS REFLECTION_ORIGINAL_CHAINS)
        list(FIND PS2X_VU1_STATIC_SOURCES "${CHAIN}" REFLECTION_CHAIN_INDEX)
        if(REFLECTION_CHAIN_INDEX LESS 0)
            message(FATAL_ERROR "Reflection chain is missing from the authenticated package")
        endif()
        list(REMOVE_ITEM PS2X_VU1_STATIC_SOURCES "${CHAIN}")
    endforeach()
    if(TF_REPLAY_REFLECTION_LIVE)
        set(REPLAY_TARGET tf_reflection_live)
        add_executable(${REPLAY_TARGET} replay.cpp reflection_live.cpp "${DUAL_STATE}" "${DUAL_MATH}" "${REFLECTION_MATH}"
            "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native/tf_gpu_reflection.cpp"
            "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native/tf_reflection_memory.cpp"
            "${CMAKE_BINARY_DIR}/reflection/reflection_lighting.cpp"
            "${CMAKE_BINARY_DIR}/reflection/reflection_projection.cpp" ${PS2X_VU1_STATIC_SOURCES})
        set_source_files_properties("${DUAL_MATH}" "${REFLECTION_MATH}" PROPERTIES COMPILE_OPTIONS "/fp:strict")
        target_compile_definitions(${REPLAY_TARGET} PRIVATE TF_REPLAY_REFLECTION_LIVE=1 PS2X_VU1_GPU_REFLECTION=1)
    else()
        set(REPLAY_TARGET tf_reflection_capture)
        add_executable(${REPLAY_TARGET} replay.cpp reflection_capture.cpp "${DUAL_STATE}"
            "${CMAKE_BINARY_DIR}/reflection/reflection_chain.cpp"
            "${CMAKE_BINARY_DIR}/reflection/reflection_projection_chain.cpp" ${PS2X_VU1_STATIC_SOURCES})
        target_compile_definitions(${REPLAY_TARGET} PRIVATE TF_REPLAY_REFLECTION_CAPTURE=1)
    endif()
    target_include_directories(${REPLAY_TARGET} PRIVATE "${PACKAGE_DIR}/generated")
else()
    set(REPLAY_TARGET tf_vu1_replay)
    add_executable(${REPLAY_TARGET} replay.cpp "${DUAL_STATE}" ${PS2X_VU1_STATIC_SOURCES})
endif()
set_source_files_properties("${DUAL_STATE}" PROPERTIES COMPILE_OPTIONS "/fp:strict")
target_compile_features(${REPLAY_TARGET} PRIVATE cxx_std_20)
target_compile_definitions(${REPLAY_TARGET} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN NDEBUG)
target_compile_options(${REPLAY_TARGET} PRIVATE /O2 /Ob2 /fp:precise /arch:AVX2)
target_include_directories(${REPLAY_TARGET} PRIVATE
    "${REPO_ROOT}/PS2Recomp/ps2xRuntime/src/lib/native"
    "${REPO_ROOT}/tools/vu1/toolchain/shim"
    "${PACKAGE_DIR}/native/include"
    "${PACKAGE_DIR}/native/src/lib/vu"
    "${REPO_ROOT}/PS2Recomp/ps2xRuntime/include")
